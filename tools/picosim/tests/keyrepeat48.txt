# Held keys repeat where something is typed (as the Spectrum's own keys do: after 0.7 s, then
# every 0.1 s), and nowhere else: the menu's own boxes (Find), a text file's editor, the notes
# and details of a game (M on its information). Paging a text file with N, or a game's
# information, does not repeat. ENTER and BREAK never repeat.
sd /48.rom {ROM48}
sd /Docs/long.txt {HERE}/long.txt
sd /Docs/short.txt {HERE}/short.txt
sd /Games/Green.tap {HERE}/tapein.tap
sd /Game Notes/readme.txt {HERE}/short.txt
boot
nmi
wait 30
key b
wait 50
# Find (the ROM's own box)
key f
wait 30
key g
hold r 75
wait 5
expect grrrr
hold DELETE 100
wait 5
expectnot grr
key x
wait 5
expect x_
key BREAK
wait 30
# a text file: N held turns one page; in the editor a held letter repeats
key 1
wait 50
key l
wait 50
expect 1-long
key 1
wait 50
expect Pg 1
hold n 100
wait 10
expect Pg 2
expectnot Pg 3
key x
wait 50
key 2
wait 50
key e
wait 30
expect Edit: short.txt
hold a 75
wait 10
expect aaaa
key BREAK
wait 30
expect Y-Save
key n
wait 30
key x
wait 50

key 1
