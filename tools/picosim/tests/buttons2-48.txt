# Picomenu.ini (tests/buttons-b.ini): save_press = double swaps them, so a double press is the
# save screen (no game: the Save snapshot screen) and a hold the screen grab question. An old
# long_press = 1.0 is read as 3 seconds (under 2 would leave no time for the hold), and the old
# double_press = grab as on. Options shows them; double_press = off makes two presses the menu.
sd /48.rom {ROM48}
sd /Picomenu.ini {HERE}/buttons-b.ini
boot
button double 120 150 120
wait 30
expect Save Snapshot
expectnot B-Browse
key BREAK
wait 30
expectnot Save Snapshot
expectnot B-Browse
button hold 1500
wait 30
expect Save it as:
key BREAK
wait 30
expectnot Save it as:
button hold 2500
wait 30
expect Save it as:
key BREAK
wait 30
