# Picomenu.ini (tests/slots-b.ini): save_screen = named, so S is the Save snapshot screen
# straight away; slots_on_load = never, so a game with slots loads with no question. Q still
# saves over the slot used last.
sd /48.rom {ROM48}
sd /Picomenu.ini {HERE}/slots-b.ini
sd /Games/Alpha.z80 {TAPES}/Green_Beret_1986ImagineSpeedLock_2-48.z80
sd /Save Games/Alpha/Slot 3 - old.z80 {TAPES}/Green_Beret_1986ImagineSpeedLock_2-48.z80
sd /Save Games/Alpha/.last3/keep {HERE}/slots-last3.dat
boot
nmi
wait 30
key b
wait 50
key 1
wait 50
key 1
wait 150
expectnot Saved games
nmi
wait 30
expect Reload: Alpha
key s
wait 30
expect Save Snapshot
key BREAK
wait 30
key q
wait 100
nofile sd /Save Games/Alpha Save Game.z80
get sd /Save Games/Alpha/Slot 3 - old.sna {OUT}/s3.sna
nofile sd /Save Games/Alpha/Slot 3 - old.z80
