# Picomenu.ini (tests/slots-b.ini): save_screen = named, so S is the Save snapshot screen
# straight away (and from the button held a second); slots_on_load = never, so a game with slots
# loads with no question.
sd /48.rom {ROM48}
sd /Picomenu.ini {HERE}/slots-b.ini
sd /Games/Alpha.z80 {TAPES}/Green_Beret_1986ImagineSpeedLock_2-48.z80
sd /Save Games/Alpha/Slot 3 - old.z80 {TAPES}/Green_Beret_1986ImagineSpeedLock_2-48.z80
sd /Save Games/Alpha/.last3/keep {HERE}/slots-last3.dat
boot
nmi
wait 30
key b
wait 50
key 1
wait 50
key 1
wait 150
expectnot Saved games
nmi
wait 30
expect Reload: Alpha
key s
wait 30
expect Save Snapshot
key BREAK
wait 30
expect B-Browse
key x
wait 30
# the button held a second: the Save snapshot screen too, with no menu; BREAK goes back to the game
nmihold
wait 30
expect Save Snapshot
expectnot B-Browse
key BREAK
wait 30
expectnot Save Snapshot
expectnot B-Browse
