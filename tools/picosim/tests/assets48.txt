# E in Tools: Erase all assets. Part of a name typed lists the games with assets (from the
# folders they are kept in); one picked shows its assets by category, with how many: its
# autosaves, save slots and quick saves (in /Save Games, the save folder, the last game's), notes and details, pokes, tape
# notes and labels, screen grabs. 1-6 turn categories off and on; E, then Y, erases what is on
# (to the Waste Bin). Another game's whose name starts the same is not touched.
sd /48.rom {ROM48}
sd /Games/Exolon.z80 {OUT}/dummy.scr
sd /Save Games/Exolon Save Game.z80 {OUT}/dummy.scr
sd /Save Games/Exolon Save Game 2.sna {OUT}/dummy.scr
sd /Save Games/Exolonia Save Game.z80 {OUT}/dummy.scr
sd /Games/Old/Exolon Save Game 3.sna {OUT}/dummy.scr
sd /Autosave/Exolon/Autosave 1.z80 {OUT}/dummy.scr
sd /Save Games/Exolon/Slot 1.z80 {OUT}/dummy.scr
sd /Game Notes/Exolon.txt {HERE}/short.txt
sd /Game Notes/Exolon.info.txt {HERE}/short.txt
sd /Poke List/Exolon.pok {OUT}/test.pok
sd /Tape Files/Games~Exolon.z80.tape {HERE}/short.txt
sd /Tape Files/Games~Exolon.tap.labels {HERE}/short.txt
sd /Scraper/Screen Grabs/Exolon Loading Screen.scr {OUT}/dummy.scr
sd /Scraper/Screen Grabs/Exolon1.scr {OUT}/dummy.scr
sd /Scraper/Screen Grabs/Exolon Game Screen2.scr {OUT}/dummy.scr
sd /Scraper/Screen Grabs/Exolonia.scr {OUT}/dummy.scr
boot
nmi
wait 30
key t
wait 30
expect E-Erase all assets
key e
wait 30
expect The game (part of its name):
key e
key x
key o
wait 10
expect exo_
key ENTER
wait 60
expect 2 games with assets
expect 1 Exolon
expect 2 Exolonia
key 1
wait 60
expect 1 [x] Autosaves
expect 2 [x] Save slots, saves   3
expect 3 [x] Game Notes          2
expect 4 [x] Pokes (.pok)        1
expect 5 [x] Tape files          2
expect 6 [x] Screen grabs        3
key 6
wait 20
expect 6 [ ] Screen grabs        3
key e
wait 20
expect Erase 9? Y-Yes
key y
wait 100
expect Erased 9 (to the Waste Bin)
nofile sd /Autosave/Exolon
nofile sd /Save Games/Exolon
nofile sd /Save Games/Exolon Save Game.z80
nofile sd /Save Games/Exolon Save Game 2.sna
nofile sd /Game Notes/Exolon.txt
nofile sd /Game Notes/Exolon.info.txt
nofile sd /Poke List/Exolon.pok
nofile sd /Tape Files/Games~Exolon.z80.tape
nofile sd /Tape Files/Games~Exolon.tap.labels
expectbyte sd /Games/Exolon.z80|0|0x55
expectbyte sd /Save Games/Exolonia Save Game.z80|0|0x55
expectbyte sd /Scraper/Screen Grabs/Exolon1.scr|0|0x55
expect 6 [x] Screen grabs        3
# now the grabs only
key a
wait 20
key e
key y
wait 60
expect Erased 3 (to the Waste Bin)
nofile sd /Scraper/Screen Grabs/Exolon Loading Screen.scr
nofile sd /Scraper/Screen Grabs/Exolon1.scr
nofile sd /Scraper/Screen Grabs/Exolon Game Screen2.scr
expectbyte sd /Scraper/Screen Grabs/Exolonia.scr|0|0x55
key a
wait 20
expect Nothing to erase
key BREAK
wait 40
expect 1 game with assets
expect 1 Exolonia
key BREAK
wait 20
key BREAK
wait 30
expect E-Erase all assets
