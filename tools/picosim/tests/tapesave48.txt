# Recording (W on the tape position screen): the tape is named, then SAVE "c" CODE 30000,4 in
# 48 BASIC goes into /Recordings/mytape.tap at once (header and data, as a .tap). E stops it; the
# name is then refused as in use. The recording is put in with T and LOAD ""CODE gets the bytes
# back.
flash /48.rom {ROM48}
boot
nmi
wait 30
key v
wait 40
expect W-Record
key w
wait 30
expect Record a new tape
key m
key y
key t
key a
key p
key e
wait 20
expect mytape_.tap
key ENTER
wait 40
expect Recording
expect Nothing saved on it yet
key x
wait 30
key x
wait 30

key 3

key 3
key 0
key 0
key 0
key 0

key 4
key ENTER

key a
wait 200

wait 40
