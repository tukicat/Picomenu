# ({TAPES}/picomenu-test.uf2: any PicoMenu .uf2, kept with the tapes, not in the source)
# T-U, Update PicoMenu, from a .uf2 in the root of the SD card: copied to the internal flash,
# then written (on the PC the firmware is an array), and the done screen's lines from the Pico:
# with the reset mod (reset_pin = 28), the Pico resets the Spectrum itself
sd /48.rom {ROM48}
sd /Picomenu.ini {HERE}/reset-pin.ini
sd /picomenu.uf2 {TAPES}/picomenu-test.uf2
boot
nmi
wait 30
key t
wait 30
key u
wait 200
expect Update PicoMenu
key y
wait 3000
expect Update complete
expect Wait for automatic restart.
